--NH Voalte

USE [identityiq];
GO
CREATE SCHEMA igacustom;
GO

CREATE TABLE igacustom.NH_VOALTE_EXTENSION (
    EXTENSION        VARCHAR(15)  NOT NULL PRIMARY KEY,
    BUILDING_NAME    VARCHAR(80)  NOT NULL,
    STATUS           VARCHAR(12)  NOT NULL,
 
    CONSTRAINT CK_NH_VOALTE_STATUS
        CHECK (STATUS IN ('Available', 'InProgress', 'Used'))
);
GO
 
CREATE NONCLUSTERED INDEX IDX_NH_VOALTE_BUILDING_STATUS
ON igacustom.NH_VOALTE_EXTENSION (BUILDING_NAME, STATUS);
GO

--NHRMC Voalte

CREATE TABLE igacustom.NHRMC_VOALTE_EXTENSION (
    EXTENSION       VARCHAR(15)  NOT NULL PRIMARY KEY,
    BUILDING_ID     VARCHAR(12)  NOT NULL,
    BUILDING_NAME   VARCHAR(80) NOT NULL,
    STATUS          VARCHAR(12)  NOT NULL,

    CONSTRAINT CK_NHRMC_VOALTE_STATUS
        CHECK (STATUS IN ('Assigned', 'Unassigned'))
);
 
GO

CREATE NONCLUSTERED INDEX IDX_NHRMC_VOALTE_BUILDING_STATUS
ON igacustom.NHRMC_VOALTE_EXTENSION (BUILDING_ID, STATUS);
GO
